/**
 * Sirve la página web principal, pagos o cuenta corriente según el parámetro en la URL.
 */
function doGet(e) {
  var pagina = 'index'; 
  
  if (e && e.parameter) {
    if (e.parameter.page === 'pagos') pagina = 'pagos';
    if (e.parameter.page === 'cuenta') pagina = 'cuenta';
  }
  
  return HtmlService.createTemplateFromFile(pagina)
      .evaluate()
      .setTitle('Puffco Mayorista')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
      .addMetaTag('viewport', 'width=device-width, initial-scale=1, maximum-scale=1');
}

function obtenerDatos() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheets()[0]; 
    
    var fila_inicio = 2; 
    var ultimaFila = sheet.getLastRow();
    
    if (ultimaFila < fila_inicio) return [];

    var rango = sheet.getRange(fila_inicio, 1, ultimaFila - fila_inicio + 1, 8);
    var datos = rango.getValues();

    var productos = datos.map(function(row) {
      if (!row[0]) return null;
      return {
        nombre: row[0],     
        precio: row[1],     
        stock: row[3],      
        marca: row[5] ? row[5].toString().toUpperCase() : "VARIOS",    
        categoria: row[6] ? row[6].toString().toUpperCase() : "GENERAL" 
      };
    }).filter(function(p) { return p !== null; });

    productos.sort(function(a, b) {
      var esPuffcoA = a.marca.includes("PUFFCO");
      var esPuffcoB = b.marca.includes("PUFFCO");
      if (esPuffcoA && !esPuffcoB) return -1;
      if (!esPuffcoA && esPuffcoB) return 1;
      if (a.marca < b.marca) return -1;
      if (a.marca > b.marca) return 1;
      return 0;
    });
    
    return productos;
  } catch (e) {
    throw new Error("Error: " + e.message);
  }
}

function obtenerClientes() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var hojas = ss.getSheets();
  var nombresClientes = [];
  
  var ignorar = ['Resumen', 'Stock y price', 'LISTA Special distro', 'Pedidos', 'Control de errores', 'Costo Carga', 'PESOS', 'COMPRAS Proveedores', 'Cta Cte total'];
  
  for (var i = 0; i < hojas.length; i++) {
    var nombre = hojas[i].getName();
    if (ignorar.indexOf(nombre) === -1) {
      nombresClientes.push(nombre);
    }
  }
  
  nombresClientes.sort();
  return nombresClientes;
}

function procesarPedido(info) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); 
  } catch (e) {
    throw new Error('Servidor ocupado, intenta de nuevo.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaStock = ss.getSheets()[0];
    var dataStock = hojaStock.getDataRange().getValues();
    var gananciaTotal = 0;
    
    for (var k = 0; k < info.items.length; k++) {
      var itemCompra = info.items[k];
      
      for (var i = 1; i < dataStock.length; i++) {
        if (dataStock[i][0] == itemCompra.nombre) {
          var stockActual = parseInt(dataStock[i][3]); 
          if (isNaN(stockActual)) stockActual = 0;
          
          var nuevoStock = stockActual - itemCompra.cantidad;
          if (nuevoStock < 0) nuevoStock = 0; 
          
          hojaStock.getRange(i + 1, 4).setValue(nuevoStock);
          
          var precioVenta = limpiarNumero(dataStock[i][1]); 
          var costoProducto = limpiarNumero(dataStock[i][7]); 
          
          var gananciaUnitaria = precioVenta - costoProducto;
          gananciaTotal += (gananciaUnitaria * itemCompra.cantidad);
          
          break; 
        }
      }
    }

    var hojaPedidos = ss.getSheetByName('Pedidos');
    if (!hojaPedidos) {
      hojaPedidos = ss.insertSheet('Pedidos');
      hojaPedidos.appendRow(['Fecha', 'Cliente', 'Detalle', 'Total', 'Ganancia']);
    }
    
    var filas = hojaPedidos.getRange("B:B").getValues();
    var ultimaFilaConDatos = 0;
    for (var i = filas.length - 1; i >= 0; i--) {
      if (filas[i][0] !== "") {
        ultimaFilaConDatos = i + 1;
        break;
      }
    }
    
    var filaDestino = (ultimaFilaConDatos === 0) ? 2 : ultimaFilaConDatos + 1;
    var valoresAInsertar = [new Date(), info.cliente, info.detalle, info.total, gananciaTotal];
    hojaPedidos.getRange(filaDestino, 1, 1, 5).setValues([valoresAInsertar]);
    
    SpreadsheetApp.flush();
    enviarEmailNotificacion(info, gananciaTotal);

    return "OK";
  } catch (e) {
    throw new Error("Error procesando: " + e.message);
  } finally {
    lock.releaseLock();
  }
}

function limpiarNumero(valor) {
  if (!valor) return 0;
  if (typeof valor === 'number') return valor;
  var limpio = valor.toString().replace(',', '.');
  var numero = parseFloat(limpio);
  return isNaN(numero) ? 0 : numero;
}

function registrarPago(datosPago) {
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); 
  } catch (e) {
    throw new Error('El sistema está ocupado, intentá en unos segundos.');
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hojaCliente = ss.getSheetByName(datosPago.cliente);
    
    if (!hojaCliente) {
      throw new Error("No se encontró la pestaña del cliente: " + datosPago.cliente);
    }
    
    var pagoEnUSD = 0;
    var observacion = "";
    
    if (datosPago.moneda === "USD") {
      pagoEnUSD = parseFloat(datosPago.monto);
      observacion = "Pago en Efectivo/Transf USD";
    } else { 
      var montoPesos = parseFloat(datosPago.monto);
      var cotizacion = parseFloat(datosPago.cotizacion);
      pagoEnUSD = montoPesos / cotizacion;
      observacion = "Pagó $" + montoPesos + " ARS (Cotiz: $" + cotizacion + ") - " + datosPago.detalle;
    }
    
    var columnaFechas = hojaCliente.getRange("E:E").getValues();
    var filaDestino = 0;
    
    for (var i = 1; i < columnaFechas.length; i++) {
      if (columnaFechas[i][0] === "") {
        filaDestino = i + 1; 
        break;
      }
    }
    
    if (filaDestino === 0) {
      filaDestino = hojaCliente.getLastRow() + 1;
    }
    
    var fechaHoy = new Date();
    
    hojaCliente.getRange(filaDestino, 5).setValue(fechaHoy);
    hojaCliente.getRange(filaDestino, 6).setValue(pagoEnUSD);
    hojaCliente.getRange(filaDestino, 11).setValue(observacion);
    
    SpreadsheetApp.flush();
    return "OK";
    
  } catch (e) {
    throw new Error("Error al asentar pago: " + e.message);
  } finally {
    lock.releaseLock();
  }
}

/**
 * Lee la pestaña del cliente, mezcla compras (A-D) y pagos (E-K) 
 * y devuelve un historial ordenado por fecha.
 */
function obtenerEstadoCuenta(cliente) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var hoja = ss.getSheetByName(cliente);
    if (!hoja) throw new Error("Cliente no encontrado.");
    
    // Leemos todo el bloque de datos a partir de la fila 3 (evitando los encabezados y saldos de arriba)
    var ultimaFila = hoja.getLastRow();
    if (ultimaFila < 3) return { movimientos: [], saldoTotal: 0 };
    
    // Traemos de la A (1) hasta la K (11)
    var datos = hoja.getRange(3, 1, ultimaFila - 2, 11).getValues();
    var movimientos = [];
    
    for (var i = 0; i < datos.length; i++) {
      var fila = datos[i];
      
      // 1. Extraer Pedidos (Columnas A, B, C, D -> índices 0, 1, 2, 3)
      var fechaPedido = fila[0];
      var aRendir = limpiarNumero(fila[3]); // Columna D (A rendir)
      
      if (fechaPedido && fechaPedido instanceof Date && aRendir > 0) {
        movimientos.push({
          fecha: fechaPedido,
          tipo: 'Cargo',
          detalle: 'Pedido',
          cargo: aRendir,
          abono: 0
        });
      }
      
      // 2. Extraer Pagos (Columnas E, F, K -> índices 4, 5, 10)
      var fechaPago = fila[4];
      var pagoAbonado = limpiarNumero(fila[5]); // Columna F (Pago USD)
      var observaciones = fila[10] ? fila[10].toString() : "Pago ingresado";
      
      if (fechaPago && fechaPago instanceof Date && pagoAbonado > 0) {
        movimientos.push({
          fecha: fechaPago,
          tipo: 'Abono',
          detalle: observaciones,
          cargo: 0,
          abono: pagoAbonado
        });
      }
    }
    
    // Ordenar todo cronológicamente por fecha (del más viejo al más nuevo)
    movimientos.sort(function(a, b) {
      return a.fecha.getTime() - b.fecha.getTime();
    });
    
    // Calcular el saldo acumulado renglón por renglón
    var saldoAcumulado = 0;
    var historial = movimientos.map(function(mov) {
      saldoAcumulado = saldoAcumulado + mov.cargo - mov.abono;
      
      // Formatear la fecha para que se vea linda (DD/MM/YYYY)
      var d = mov.fecha;
      var fechaStr = ("0" + d.getDate()).slice(-2) + "/" + ("0" + (d.getMonth() + 1)).slice(-2) + "/" + d.getFullYear();
      
      return {
        fecha: fechaStr,
        detalle: mov.detalle,
        cargo: mov.cargo,
        abono: mov.abono,
        saldo: saldoAcumulado
      };
    });
    
    // Invertimos el array para que lo más nuevo se vea arriba en la tabla
    historial.reverse();
    
    return {
      movimientos: historial,
      saldoTotal: saldoAcumulado
    };
    
  } catch (e) {
    throw new Error(e.message);
  }
}
