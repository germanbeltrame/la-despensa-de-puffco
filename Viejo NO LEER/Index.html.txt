<script>
  var productosGlobal = [];
  var carrito = [];

  window.onload = function() {
    google.script.run
      .withSuccessHandler(llenarSelectClientes)
      .withFailureHandler(function(e){ console.error("Error cargando clientes: " + e); })
      .obtenerClientes();

    google.script.run
      .withSuccessHandler(mostrarProductos)
      .withFailureHandler(function(e){ alert("Error: " + e); })
      .obtenerDatos();
  };

  function llenarSelectClientes(clientes) {
    var select = document.getElementById('select-cliente');
    select.innerHTML = '<option value="">-- Selecciona tu nombre --</option>';
    
    clientes.forEach(function(cliente) {
      select.innerHTML += '<option value="' + cliente + '">' + cliente + '</option>';
    });
    
    select.innerHTML += '<option value="NUEVO">➕ OTRO / NUEVO CLIENTE (Contado)</option>';

    var nombreGuardado = localStorage.getItem('nombreClientePuffco');
    if (nombreGuardado) {
      if (clientes.includes(nombreGuardado)) {
        select.value = nombreGuardado; 
      } else {
        select.value = "NUEVO"; 
        document.getElementById('input-cliente').value = nombreGuardado;
        document.getElementById('input-cliente').style.display = 'block';
      }
    }
  }

  function verificarNuevoCliente() {
    var select = document.getElementById('select-cliente');
    var input = document.getElementById('input-cliente');
    
    if (select.value === "NUEVO") {
      input.style.display = "block";
      input.focus();
    } else {
      input.style.display = "none";
      if (select.value !== "") {
         localStorage.setItem('nombreClientePuffco', select.value);
      }
    }
  }

  function guardarNombre() {
    var input = document.getElementById('input-cliente').value;
    localStorage.setItem('nombreClientePuffco', input);
  }

  function mostrarProductos(productos) {
    productosGlobal = productos;
    var div = document.getElementById('catalogo');
    var loader = document.getElementById('loader');
    loader.style.display = 'none';
    div.innerHTML = "";

    if (productos.length === 0) {
      div.innerHTML = "<p>No hay productos disponibles o stock cargado.</p>";
      return;
    }

    productos.forEach(function(p, index) {
      var tieneStock = p.stock > 0;
      var claseStock = tieneStock ? "stock" : "sin-stock";
      var textoStock = tieneStock ? "Stock: " + p.stock : "Sin Stock";
      
      var html = `
        <div class="producto-card" data-texto="${p.nombre} ${p.marca} ${p.categoria}">
          <div class="info-prod">
            <h3>${p.nombre}</h3>
            <div class="etiquetas">
              <span class="tag">${p.marca}</span>
              <span class="tag" style="background-color: #e3f2fd; color: #1565c0;">${p.categoria}</span>
            </div>
          </div>
          
          <div class="precio-stock">
            <span class="precio">$${p.precio}</span>
            <span class="${claseStock}">${textoStock}</span>
          </div>

          <div class="acciones">
            <input type="number" id="cant-${index}" class="cantidad" min="1" max="${p.stock}" value="1" ${!tieneStock ? 'disabled' : ''}>
            <button class="btn-agregar" onclick="agregarAlCarrito(${index})" ${!tieneStock ? 'disabled' : ''}>
              ${tieneStock ? 'Agregar' : 'Agotado'}
            </button>
          </div>
        </div>
      `;
      div.innerHTML += html;
    });
  }

  function filtrarProductos() {
    var input = document.getElementById('buscador').value.toUpperCase();
    var cards = document.getElementsByClassName('producto-card');
    
    for (var i = 0; i < cards.length; i++) {
      var texto = cards[i].getAttribute('data-texto').toUpperCase();
      cards[i].style.display = texto.includes(input) ? "" : "none";
    }
  }

  function agregarAlCarrito(index) {
    var p = productosGlobal[index];
    var cantInput = document.getElementById('cant-' + index);
    var cantidad = parseInt(cantInput.value);
    
    if (cantidad > p.stock) {
      alert("No hay suficiente stock. Disponible: " + p.stock);
      return;
    }
    if (cantidad <= 0) return;

    var existente = carrito.find(item => item.nombre === p.nombre);
    if (existente) {
      if (existente.cantidad + cantidad > p.stock) {
         alert("Límite de stock alcanzado en el carrito.");
         return;
      }
      existente.cantidad += cantidad;
    } else {
      carrito.push({
        nombre: p.nombre,
        precio: p.precio,
        cantidad: cantidad
      });
    }
    
    actualizarUIcarrito();
    cantInput.value = 1;
    
    var btn = document.querySelectorAll('.btn-agregar')[index];
    var textoOriginal = btn.innerText;
    btn.innerText = "¡Listo!";
    setTimeout(() => btn.innerText = textoOriginal, 1000);
  }

  function actualizarUIcarrito() {
    var btnFloat = document.getElementById('carrito-flotante');
    var count = document.getElementById('count-carrito');
    var totalItems = carrito.reduce((a, b) => a + b.cantidad, 0);
    
    count.innerText = totalItems;
    btnFloat.style.display = totalItems > 0 ? 'block' : 'none';
  }

  function abrirCarrito() {
    var select = document.getElementById('select-cliente');
    var input = document.getElementById('input-cliente');
    var nombre = select.value === "NUEVO" ? input.value : select.value;
    
    var modal = document.getElementById('modalCarrito');
    var lista = document.getElementById('lista-pedido');
    var spanTotal = document.getElementById('total-pedido');
    var spanNombre = document.getElementById('nombre-resumen');

    spanNombre.innerText = nombre || "--- (Falta completar nombre arriba) ---";
    spanNombre.style.color = nombre ? "black" : "red";
    
    lista.innerHTML = "";
    var total = 0;
    
    carrito.forEach((item, idx) => {
      var subtotal = item.precio * item.cantidad;
      total += subtotal;
      lista.innerHTML += `
        <li>
          <span>${item.cantidad} x ${item.nombre}</span>
          <div>
            <span>$${subtotal}</span>
            <a href="#" onclick="borrarItem(${idx})" style="color:red;margin-left:10px;text-decoration:none;font-weight:bold;">✕</a>
          </div>
        </li>`;
    });
    
    spanTotal.innerText = total;
    modal.style.display = "block";
  }

  function borrarItem(idx) {
    carrito.splice(idx, 1);
    abrirCarrito(); 
    actualizarUIcarrito();
    if(carrito.length === 0) cerrarCarrito();
  }

  function cerrarCarrito() {
    document.getElementById('modalCarrito').style.display = "none";
  }

  function enviarPedido() {
    var select = document.getElementById('select-cliente');
    var input = document.getElementById('input-cliente');
    var cliente = "";

    if (select.value === "NUEVO") {
      cliente = input.value.trim();
    } else {
      cliente = select.value;
    }

    if (!cliente) {
      alert("⚠️ FALTAN DATOS:\nPor favor selecciona un cliente de la lista o escribe el nombre.");
      cerrarCarrito();
      select.focus();
      window.scrollTo(0, 0);
      return;
    }
    
    if (carrito.length === 0) return;

    var btn = document.getElementById('btn-confirmar');
    btn.disabled = true;
    btn.innerText = "Procesando...";

    var detalleTexto = carrito.map(i => `${i.cantidad} x ${i.nombre}`).join(", ");
    var total = carrito.reduce((a, b) => a + (b.precio * b.cantidad), 0);

    var pedidoInfo = {
      cliente: cliente,
      detalle: detalleTexto,
      total: total,
      items: carrito
    };

    google.script.run
      .withSuccessHandler(function(res) {
        if (res === "OK") {
          alert("✅ ¡Pedido realizado con éxito!\nGracias " + cliente);
          carrito = [];
          actualizarUIcarrito();
          cerrarCarrito();
          location.reload(); 
        }
      })
      .withFailureHandler(function(e) {
        alert("Error: " + e.message);
        btn.disabled = false;
        btn.innerText = "Confirmar Pedido";
      })
      .procesarPedido(pedidoInfo);
  }
</script>
