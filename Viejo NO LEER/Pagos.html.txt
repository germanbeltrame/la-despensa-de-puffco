<script>
  var monedaActual = 'USD';

  window.onload = function() {
    google.script.run
      .withSuccessHandler(function(clientes) {
        var select = document.getElementById('select-cliente');
        select.innerHTML = '<option value="">-- Elegir cliente --</option>';
        clientes.forEach(c => select.innerHTML += '<option value="'+c+'">'+c+'</option>');
      })
      .obtenerClientes();

    // Event listeners para calcular preview en vivo
    document.getElementById('monto-ars').addEventListener('input', calcularPreview);
    document.getElementById('cotiz-ars').addEventListener('input', calcularPreview);
  };

  function cambiarMoneda(moneda) {
    monedaActual = moneda;
    document.getElementById('btn-usd').className = moneda === 'USD' ? 'moneda-btn active' : 'moneda-btn';
    document.getElementById('btn-ars').className = moneda === 'ARS' ? 'moneda-btn active' : 'moneda-btn';
    
    document.getElementById('usd-group').style.display = moneda === 'USD' ? 'block' : 'none';
    document.getElementById('pesos-group').style.display = moneda === 'ARS' ? 'block' : 'none';
  }

  function calcularPreview() {
    var pesos = parseFloat(document.getElementById('monto-ars').value) || 0;
    var cotiz = parseFloat(document.getElementById('cotiz-ars').value) || 0;
    var preview = document.getElementById('preview-usd');
    
    if (pesos > 0 && cotiz > 0) {
      var dolares = (pesos / cotiz).toFixed(2);
      preview.innerText = "$" + dolares + " USD";
    } else {
      preview.innerText = "$0.00 USD";
    }
  }

  function enviarPago() {
    var cliente = document.getElementById('select-cliente').value;
    if (!cliente) { alert("Elegí un cliente."); return; }

    var datosPago = { cliente: cliente, moneda: monedaActual };

    if (monedaActual === 'USD') {
      var montoUsd = parseFloat(document.getElementById('monto-usd').value);
      if (!montoUsd || montoUsd <= 0) { alert("Ingresá un monto válido en dólares."); return; }
      datosPago.monto = montoUsd;
    } else {
      var montoArs = parseFloat(document.getElementById('monto-ars').value);
      var cotiz = parseFloat(document.getElementById('cotiz-ars').value);
      var detalle = document.getElementById('detalle-ars').value || "-";
      
      if (!montoArs || montoArs <= 0 || !cotiz || cotiz <= 0) {
        alert("Completá los Pesos y la Cotización correctamente."); return;
      }
      datosPago.monto = montoArs;
      datosPago.cotizacion = cotiz;
      datosPago.detalle = detalle;
    }

    if (!confirm("¿Asentar este pago en la cuenta de " + cliente + "?")) return;

    var btn = document.getElementById('btn-guardar');
    btn.disabled = true;
    btn.innerText = "Guardando...";

    google.script.run
      .withSuccessHandler(function(res) {
        if (res === "OK") {
          alert("✅ Pago registrado en la cuenta de " + cliente);
          location.reload();
        }
      })
      .withFailureHandler(function(err) {
        alert("Error: " + err.message);
        btn.disabled = false;
        btn.innerText = "Asentar Pago en Excel";
      })
      .registrarPago(datosPago);
  }
</script>
