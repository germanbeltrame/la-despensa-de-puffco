<script>
  window.onload = function() {
    google.script.run
      .withSuccessHandler(function(clientes) {
        var select = document.getElementById('select-cliente');
        select.innerHTML = '<option value="">-- Elegir cliente --</option>';
        clientes.forEach(c => select.innerHTML += '<option value="'+c+'">'+c+'</option>');
      })
      .obtenerClientes();
  };

  function consultarCuenta() {
    var cliente = document.getElementById('select-cliente').value;
    if (!cliente) {
      alert("Por favor elegí un cliente de la lista.");
      return;
    }

    // Resetear y mostrar cargando
    document.getElementById('btn-consultar').disabled = true;
    document.getElementById('loader').style.display = 'block';
    document.getElementById('tabla-container').style.display = 'none';
    document.getElementById('saldo-card').style.display = 'none';
    document.getElementById('tabla-body').innerHTML = '';

    google.script.run
      .withSuccessHandler(mostrarEstadoCuenta)
      .withFailureHandler(function(err) {
        alert("Error leyendo el Excel: " + err.message);
        document.getElementById('btn-consultar').disabled = false;
        document.getElementById('loader').style.display = 'none';
      })
      .obtenerEstadoCuenta(cliente);
  }

  function mostrarEstadoCuenta(datos) {
    document.getElementById('btn-consultar').disabled = false;
    document.getElementById('loader').style.display = 'none';
    
    // 1. Mostrar la tarjeta de saldo total
    var saldoCard = document.getElementById('saldo-card');
    var saldoTotalEl = document.getElementById('saldo-total');
    
    saldoCard.style.display = 'block';
    
    var saldoNum = parseFloat(datos.saldoTotal);
    saldoTotalEl.innerText = "$" + formatMoney(Math.abs(saldoNum)); // Muestra el valor absoluto
    
    // Aplicar color según si debe plata o tiene a favor
    if (saldoNum > 0.01) {
      saldoTotalEl.className = 'saldo-monto deuda';
      saldoTotalEl.innerText += " (En contra)";
    } else if (saldoNum < -0.01) {
      saldoTotalEl.className = 'saldo-monto a-favor';
      saldoTotalEl.innerText += " (A favor)";
    } else {
      saldoTotalEl.className = 'saldo-monto';
      saldoTotalEl.innerText = "$0.00 (Al día)";
    }

    // 2. Llenar la tabla renglón por renglón
    var tbody = document.getElementById('tabla-body');
    if (datos.movimientos.length === 0) {
      tbody.innerHTML = '<tr><td colspan="5" style="text-align:center; padding: 20px;">No hay movimientos registrados para este cliente.</td></tr>';
    } else {
      var filasHtml = '';
      datos.movimientos.forEach(function(mov) {
        var cargoStr = mov.cargo > 0 ? '<span class="cargo">$' + formatMoney(mov.cargo) + '</span>' : '-';
        var abonoStr = mov.abono > 0 ? '<span class="abono">$' + formatMoney(mov.abono) + '</span>' : '-';
        
        filasHtml += `
          <tr>
            <td>${mov.fecha}</td>
            <td>${mov.detalle}</td>
            <td>${cargoStr}</td>
            <td>${abonoStr}</td>
            <td style="font-weight:bold;">$${formatMoney(mov.saldo)}</td>
          </tr>
        `;
      });
      tbody.innerHTML = filasHtml;
    }
    
    document.getElementById('tabla-container').style.display = 'block';
  }

  // Función para que los números se vean lindos (ej. 1.500,50)
  function formatMoney(amount) {
    return Number(amount).toLocaleString('es-AR', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  }
</script>
